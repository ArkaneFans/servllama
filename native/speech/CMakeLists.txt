cmake_minimum_required(VERSION 3.22)
project(servllama_speech LANGUAGES C CXX)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "ServLlama speech bundle supports Android arm64-v8a")
endif()
if(NOT EXISTS "${CRISPASR_SOURCE}/src/crispasr.cpp")
  message(FATAL_ERROR "Set CRISPASR_SOURCE to the pinned CrispASR checkout")
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_compile_definitions(GGML_MAX_NAME=128)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(CMAKE_DISABLE_FIND_PACKAGE_OpenMP ON CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF)
add_subdirectory("${CRISPASR_SOURCE}/ggml" ggml)
set(BUILD_SHARED_LIBS ON)
set(CRISPASR_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(CRISPASR_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CRISPASR_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(CRISPASR_CURL OFF CACHE BOOL "" FORCE)
set(CRISPASR_SDL2 OFF CACHE BOOL "" FORCE)
set(CRISPASR_OPUS OFF CACHE BOOL "" FORCE)
set(CRISPASR_AMR OFF CACHE BOOL "" FORCE)
set(CRISPASR_LIB_OUTPUT_NAME servllama_crispasr CACHE STRING "" FORCE)
add_subdirectory("${CRISPASR_SOURCE}" crispasr)
set_target_properties(crispasr-lib PROPERTIES VERSION "" SOVERSION "")
target_link_options(crispasr-lib PRIVATE
  "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map"
  "-Wl,--exclude-libs,ALL" "-Wl,-z,max-page-size=16384")
